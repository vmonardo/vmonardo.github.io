class SimpleMLP(nn.Module):
    def __init__(self, input_dim: int = 10, hidden_dim: int = 5, out_dim: int = 1):
        super().__init__()
        # First linear layer: 10 -> hidden_dim
        self.fc1 = nn.Linear(input_dim, hidden_dim)
        # Activation (fixed nonlinearity)
        self.act = nn.ReLU()
        # Second linear layer: hidden_dim -> 1 final value
        self.fc2 = nn.Linear(hidden_dim, out_dim)

    def forward(self, x):
        # x: (batch, 10)
        h = self.act(self.fc1(x))   # -> (batch, hidden_dim)
        # h2 = self.act2(self.fc2(h)) # -> (batch, hidden_dim)
        logits = self.fc2(h)       # -> (batch, 1)
        return logits

Training one epoch + evaluation¶

Pattern: model.train(), optimizer.zero_grad(), forward, loss.backward(), optimizer.step(). For evaluate: model.eval() + @torch.no_grad() (doc).

loss_fn = nn.MSELoss()
optimizer = optim.Adam(model.parameters(), lr=1e-2, weight_decay=1e-4)

def train_one_epoch(model, loader, loss_fn, optimizer, device):
    model.train()

    total_loss = 0.0
    total_seen = 0

    grad_norms = []

    for X, y in loader:
        X, y = X.to(device), y.to(device)

        optimizer.zero_grad(set_to_none=True)
        logits = model(X)
        loss = loss_fn(logits.squeeze(1), y)
        loss.backward()

        # Compute an L2 norm of all gradients (one number per batch)
        total_norm_sq = 0.0
        for p in model.parameters():
            if p.grad is not None:
                total_norm_sq += p.grad.detach().data.norm(2).item() ** 2
        grad_norms.append(math.sqrt(total_norm_sq))

        optimizer.step()

        batch_size = y.size(0)
        total_loss += loss.item() * batch_size
        total_seen += batch_size

    avg_loss = total_loss / total_seen
    avg_grad_norm = float(np.mean(grad_norms))
    return avg_loss, avg_grad_norm

@torch.no_grad()
def evaluate(model, loader, loss_fn, device):
    model.eval()

    total_loss = 0.0
    total_seen = 0

    for X, y in loader:
        X, y = X.to(device), y.to(device)
        prediction = model(X)
        loss = loss_fn(prediction.squeeze(1), y)

        batch_size = y.size(0)
        total_loss += loss.item() * batch_size
        total_seen += batch_size

    avg_loss = total_loss / total_seen
    return avg_loss

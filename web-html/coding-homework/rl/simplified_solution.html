Homework: Tabular Q-learning on the playground gridworld

You will implement the core pieces of tabular Q-learning in the same stochastic gridworld as the Q-Learning Playground demo (slip probability, terminals, negative living reward), then train and visualize the learned greedy policy.

Key ideas: (Q(s,a)) estimates discounted return; each step applies a temporal-difference backup toward $r + \gamma \max_{a'} Q(s',a')$; behavior uses $\varepsilon$-greedy exploration.

Playground gridworld (environment interface)¶

The Markov decision process (MDP) is fixed for you below: transitions (with slip), rewards, and episode termination live in step() and reset(). Skim step() so you know what (state, reward, terminated, truncated, info) means.

This is same gridworld as the one you've seen in the demos.

  • Grid $4\times 3$: columns x, rows y (origin top-left).
  • Start (0, 2). Wall (1, 1).
  • Terminal (3, 0) $\rightarrow$ +1, (3, 1) $\rightarrow$ -1.
  • Actions (order used here): Up, Right, Down, Left.
  • Slip: intended action probability 1 - 2s; each perpendicular action probability s (default s=0.1).

ACTION_NAMES = np.array(["U", "R", "D", "L"])
IDX = {label: idx for idx, label in enumerate(ACTION_NAMES)}
ACTION_DELTA = {"U": (0, -1), "R": (1, 0), "D": (0, 1), "L": (-1, 0)}
ORTHOGONAL = {"U": ("L", "R"), "R": ("U", "D"), "D": ("L", "R"), "L": ("U", "D")}


class PlaygroundGridworldEnvGymLike:
    """Stochastic gridworld equivalent to the Q-Learning Playground JS demo."""

    n_cols = 4
    n_rows = 3

    start_xy = (0, 2)
    wall_xy = (1, 1)
    good_xy = (3, 0)
    bad_xy = (3, 1)

    def __init__(self, slip: float = 0.10, living_reward: float = -0.04, max_episode_steps: int = 30, seed: int | None = 0):
        self.slip = float(slip)
        self.living_reward = float(living_reward)
        self.max_episode_steps = int(max_episode_steps)
        self.rng = np.random.default_rng(seed)

        self.terminal_rewards = {self.good_xy: 1.0, self.bad_xy: -1.0}

        self.tiles: list[tuple[int, int]] = []
        for y in range(self.n_rows):
            for x in range(self.n_cols):
                if (x, y) == self.wall_xy:
                    continue
                self.tiles.append((x, y))

        self.pos_to_idx = {pos: idx for idx, pos in enumerate(self.tiles)}
        self.n_states = len(self.tiles)
        self.n_actions = 4

        self._episode_steps = 0

    def is_terminal_xy(self, xy: tuple[int, int]) -> bool:
        return xy in self.terminal_rewards

    def clip_move(self, xy: tuple[int, int], action_label: str) -> tuple[int, int]:
        if self.is_terminal_xy(xy):
            return xy
        x, y = xy
        dx, dy = ACTION_DELTA[action_label]
        nx, ny = x + dx, y + dy
        if nx < 0 or nx >= self.n_cols or ny < 0 or ny >= self.n_rows:
            return xy
        if (nx, ny) == self.wall_xy:
            return xy
        return (nx, ny)

    def sample_executed_action(self, intended_action: int) -> int:
        """Probabilities: intent (1 - 2*slip); each perpendicular action slip."""
        label = ACTION_NAMES[int(intended_action)]
        opt_a, opt_b = ORTHOGONAL[label]
        choices = np.array([IDX[label], IDX[opt_a], IDX[opt_b]], dtype=np.int64)
        probs = np.array([1.0 - 2 * self.slip, self.slip, self.slip], dtype=float)
        pick = self.rng.choice(3, p=probs)
        return int(choices[pick])

    def reset(self):
        self._episode_steps = 0
        self.state = self.pos_to_idx[self.start_xy]
        return self.state, {}

    def step(self, action: int):
        self._episode_steps += 1
        xy = self.tiles[self.state]
        exec_a = self.sample_executed_action(action)
        next_xy = self.clip_move(xy, str(ACTION_NAMES[exec_a]))

        if self.is_terminal_xy(next_xy):
            reward = self.terminal_rewards[next_xy]
            terminated = True
        else:
            reward = self.living_reward
            terminated = False

        truncated = (not terminated) and (self._episode_steps >= self.max_episode_steps)
        self.state = self.pos_to_idx[next_xy]
        return self.state, float(reward), terminated, truncated, {"executed": int(exec_a)}

Tabular Q-learning (your implementation)¶

Update: When the next state is not terminal, $$ Q(s,a) \leftarrow Q(s,a) + \alpha \bigl(\, r + \gamma \max_{a'} Q(s',a') - Q(s,a)\,\bigr). $$ If the transition ends in a terminal state, there is no bootstrap after the absorbing reward — use $0$ for the $ \max_{a'} Q(s',\cdot)$ term (episode ends).

def greedy_action(Q: np.ndarray, state: int, rng: np.random.Generator) -> int:
    qrow = Q[state]
    maxv = np.max(qrow)
    bests = np.flatnonzero(np.isclose(qrow, maxv))
    return int(rng.choice(bests))


def train_q_learning(
    env,
    episodes: int,
    alpha: float,
    gamma: float,
    epsilon_start: float,
    epsilon_decay_per_episode: float,
    epsilon_floor: float = 0.02,
    seed: int = 0,
    env_name: str = "env",
):
    """Tabular Q-learning with exponential epsilon decay."""
    n_s = env.n_states
    n_a = env.n_actions
    Q = np.zeros((n_s, n_a), dtype=float)
    rng = np.random.default_rng(seed)

    returns = []
    successes = []
    epsilon = float(epsilon_start)

    for ep in range(episodes):
        state, _ = env.reset()
        ep_ret = 0.0
        reached_goal = False

        while True:
            if rng.random() < epsilon:
                action = int(rng.integers(n_a))
            else:
                action = greedy_action(Q, state, rng)

            next_state, reward, terminated, truncated, _ = env.step(action)
            ep_ret += reward
            done = terminated or truncated

            next_max = 0.0 if terminated else np.max(Q[next_state])
            td_target = reward + gamma * next_max
            Q[state, action] += alpha * (td_target - Q[state, action])

            state = next_state

            reached_goal |= bool(terminated and reward >= 1.0 - 1e-9)

            if done:
                break

        returns.append(ep_ret)
        successes.append(1.0 if reached_goal else 0.0)

        epsilon = max(epsilon_floor, epsilon * epsilon_decay_per_episode)

    history = {
        "env_name": env_name,
        "returns": np.array(returns),
        "successes": np.array(successes),
        "Q": Q,
    }
    return Q, history


def moving_average(x: np.ndarray, k: int) -> np.ndarray:
    if len(x) == 0:
        return x.astype(float)
    out = np.empty_like(x, dtype=float)
    c = np.cumsum(np.insert(x.astype(float), 0, 0.0))
    for i in range(len(x)):
        j = max(0, i - k + 1)
        out[i] = (c[i + 1] - c[j]) / (i - j + 1)
    return out

def plot_playground_policy(
    Q: np.ndarray,
    env: PlaygroundGridworldEnvGymLike,
    ax,
    subtitle: str,
):
    import matplotlib.patches as mpatches

    arrows = {"U": (0, -0.25), "R": (0.25, 0), "D": (0, 0.25), "L": (-0.25, 0)}

    grid = np.full((env.n_rows, env.n_cols), np.nan)
    policy = {}

    for (x, y), idx in env.pos_to_idx.items():
        if env.is_terminal_xy((x, y)):
            grid[y, x] = env.terminal_rewards[(x, y)]
        else:
            qrow = Q[idx]
            a = int(np.argmax(qrow))
            policy[(x, y)] = ACTION_NAMES[a]
            grid[y, x] = float(np.max(qrow))

    im = ax.imshow(grid, cmap="RdYlGn", vmin=-1.0, vmax=1.0)

    for y in range(env.n_rows):
        for x in range(env.n_cols):
            if np.isnan(grid[y, x]) and (x, y) == env.wall_xy:
                rect = mpatches.Rectangle(
                    (x - 0.5, y - 0.5), 1, 1, linewidth=2, edgecolor="#333", facecolor="#bbb"
                )
                ax.add_patch(rect)
                ax.text(x, y, "W", ha="center", va="center", fontsize=13, fontweight="bold")
                continue

            if env.is_terminal_xy((x, y)):
                txt = "+1" if grid[y, x] > 0 else "−1"
            else:
                txt = f"{grid[y, x]:.2f}"

            ax.text(
                x,
                y - 0.18,
                txt,
                ha="center",
                va="center",
                fontsize=9,
                color="black",
            )

            if (x, y) == env.start_xy:
                ax.text(x, y + 0.22, "S", ha="center", va="center", fontsize=9, fontweight="bold")

            if (x, y) in policy:
                gx, gy = arrows[policy[(x, y)]]
                ax.annotate(
                    "",
                    xy=(x + gx * 2, y + gy * 2),
                    xytext=(x, y),
                    arrowprops=dict(arrowstyle="->", lw=2, color="#111"),
                )

    ax.set_xticks(range(env.n_cols))
    ax.set_yticks(range(env.n_rows))
    ax.set_xticklabels(range(env.n_cols))
    ax.set_yticklabels(range(env.n_rows))
    ax.set_xlabel("x")
    ax.set_ylabel("y")
    ax.set_title(subtitle)
    return im

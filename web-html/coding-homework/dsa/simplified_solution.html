DATA_URL = "https://www.cs.toronto.edu/~kriz/cifar-10-python.tar.gz"
DATA_DIR = pathlib.Path("./").resolve()
TAR_PATH = DATA_DIR / "cifar-10-python.tar.gz"
EXTRACTED_DIR = DATA_DIR / "cifar-10-batches-py"

CAT_LABEL = 3
NEG_LABEL = 8
CAT_NAME = "Cat"
NEG_NAME = "Ship"


def download_if_needed() -> None:
    if EXTRACTED_DIR.exists():
        return
    DATA_DIR.mkdir(parents=True, exist_ok=True)
    if not TAR_PATH.exists():
        print("Downloading CIFAR-10...")
        urllib.request.urlretrieve(DATA_URL, TAR_PATH)
    print("Extracting CIFAR-10...")
    with tarfile.open(TAR_PATH, "r:gz") as tar:
        tar.extractall(path=DATA_DIR)


def _load_batch(path: pathlib.Path) -> tuple[np.ndarray, np.ndarray]:
    with path.open("rb") as handle:
        batch = pickle.load(handle, encoding="bytes")
    data = batch[b"data"]
    labels = np.array(batch[b"labels"], dtype=int)
    return data, labels


def load_data(seed: int, max_samples_per_class: int) -> tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray]:
    rng = np.random.default_rng(seed)
    batch_files = [EXTRACTED_DIR / f"data_batch_{index}" for index in range(1, 6)]
    all_data: list[np.ndarray] = []
    all_labels: list[np.ndarray] = []
    for path in batch_files:
        data, labels = _load_batch(path)
        all_data.append(data)
        all_labels.append(labels)

    X = np.vstack(all_data)
    labels = np.hstack(all_labels)
    mask = (labels == CAT_LABEL) | (labels == NEG_LABEL)
    X = X[mask]
    labels = labels[mask]
    y = (labels == CAT_LABEL).astype(int)

    idx_cat = np.where(labels == CAT_LABEL)[0]
    idx_ship = np.where(labels == NEG_LABEL)[0]
    keep_count = min(max_samples_per_class, len(idx_cat), len(idx_ship))
    idx = np.concatenate(
        [
            rng.choice(idx_cat, size=keep_count, replace=False),
            rng.choice(idx_ship, size=keep_count, replace=False),
        ]
    )
    rng.shuffle(idx)
    X = X[idx].astype(np.float32) / 255.0
    y = y[idx]

    test_batch_file = EXTRACTED_DIR / "test_batch"
    data, labels = _load_batch(test_batch_file)
    X_test = data
    labels_test = labels
    mask_test = (labels_test == CAT_LABEL) | (labels_test == NEG_LABEL)
    X_test = X_test[mask_test]
    labels_test = labels_test[mask_test]
    y_test = (labels_test == CAT_LABEL).astype(int)

    idx_cat_test = np.where(labels_test == CAT_LABEL)[0]
    idx_ship_test = np.where(labels_test == NEG_LABEL)[0]
    keep_count_test = min(max_samples_per_class, len(idx_cat_test), len(idx_ship_test))
    idx_test = np.concatenate(
        [
            rng.choice(idx_cat_test, size=keep_count_test, replace=False),
            rng.choice(idx_ship_test, size=keep_count_test, replace=False),
        ]
    )
    rng.shuffle(idx_test)
    X_test = X_test[idx_test].astype(np.float32) / 255.0
    y_test = y_test[idx_test]

    return X.reshape(-1, 3, 32, 32), y, X_test.reshape(-1, 3, 32, 32), y_test

class CatCNN(nn.Module):
    """Exactly the CNN architecture used in the referenced Kaggle notebook.

    Input:  (B, 3, 32, 32)
    Output: (B, 2) logits for binary cat vs. ship + CrossEntropyLoss
    """
    def __init__(self):
        super().__init__()

        self.layer1 = nn.Sequential(
            nn.Conv2d(in_channels=3, out_channels=32, kernel_size=3, stride=1, padding=1),
            nn.BatchNorm2d(32),
            nn.ReLU(),
            nn.MaxPool2d(kernel_size=2, stride=2),
        )

        self.layer2 = nn.Sequential(
            nn.Conv2d(in_channels=32, out_channels=64, kernel_size=3, stride=1),  # no padding
            nn.BatchNorm2d(64),
            nn.ReLU(),
            nn.MaxPool2d(kernel_size=2, stride=2),
        )

        # After layer1: (B, 32, 16, 16)
        # After layer2: Conv 3x3 w/ no padding => (B, 64, 14, 14), then MaxPool2d => (B, 64, 7, 7)
        self.fc1 = nn.Linear(in_features=64 * 7 * 7, out_features=600)
        self.drop = nn.Dropout2d(p=0.25)
        self.fc2 = nn.Linear(in_features=600, out_features=120)
        self.fc3 = nn.Linear(in_features=120, out_features=2)

    def forward(self, x):
        out = self.layer1(x)
        out = self.layer2(out)
        out = out.view(out.size(0), -1)  # flatten to (B, 2304)
        out = self.fc1(out)
        out = self.drop(out)
        out = self.fc2(out)
        out = self.fc3(out)
        return out

def accuracy_from_logits(logits: torch.Tensor, y: torch.Tensor) -> float:
    preds = logits.argmax(dim=1)
    return (preds == y).float().mean().item()

def train_one_epoch(model, loader, loss_fn, optimizer, device):
    model.train()

    total_loss = 0.0
    total_correct = 0
    total_seen = 0

    # (Optional) track gradient magnitude to build intuition
    grad_norms = []

    for X, y in loader:
        X, y = X.to(device), y.to(device)

        optimizer.zero_grad(set_to_none=True)
        logits = model(X)
        loss = loss_fn(logits, y)
        loss.backward()

        # Compute an L2 norm of all gradients (one number per batch)
        total_norm_sq = 0.0
        for p in model.parameters():
            if p.grad is not None:
                total_norm_sq += p.grad.detach().data.norm(2).item() ** 2
        grad_norms.append(math.sqrt(total_norm_sq))

        optimizer.step()

        bs = y.size(0)
        total_loss += loss.item() * bs
        total_correct += (logits.argmax(dim=1) == y).sum().item()
        total_seen += bs

    avg_loss = total_loss / total_seen
    avg_acc = total_correct / total_seen
    avg_grad_norm = float(np.mean(grad_norms))
    return avg_loss, avg_acc, avg_grad_norm

@torch.no_grad()
def evaluate(model, loader, loss_fn, device):
    model.eval()

    total_loss = 0.0
    total_correct = 0
    total_seen = 0

    for X, y in loader:
        X, y = X.to(device), y.to(device)
        logits = model(X)
        loss = loss_fn(logits, y)

        bs = y.size(0)
        total_loss += loss.item() * bs
        total_correct += (logits.argmax(dim=1) == y).sum().item()
        total_seen += bs

    avg_loss = total_loss / total_seen
    avg_acc = total_correct / total_seen
    return avg_loss, avg_acc

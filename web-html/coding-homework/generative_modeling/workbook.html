Homework Assignment: Variational Autoencoder on CIFAR-10 (binary Cat vs. Ship)

In this exercise you will build a small convolutional VAE on grayscale images, train it with the ELBO objective, and inspect reconstructions before and after training. The notebook already downloads and subsamples CIFAR-10, builds DataLoaders, and fixes hyperparameters; your work is to fill in every # TODO so the model, loss, and training loop run end-to-end.

What you will practice

  • Implementing the encoder and decoder forward passes for a convolutional VAE
  • Wiring the encoder, sampling layer, and decoder in the VAE forward
  • Writing the reconstruction and KL terms of the loss (with the given BETA weight on KL)
  • A standard PyTorch optimization step inside the training loop
  • Calling your training and validation helpers from the epoch loop

What to expect The # TODO markers tell you exactly which expressions or statements are missing. Read the inline hints next to each TODO (for example the suggested F.mse_loss shape and the KL formula). If something fails, check tensor shapes against the comments on each conv layer (spatial size halves after each stride-2 block).

Your task Complete all the to-do items in the code cells below. Do not skip the visualization cells: they help you confirm that reconstructions improve after training.

def download_if_needed() -> None:

def _load_batch(path: pathlib.Path) -> tuple[np.ndarray, np.ndarray]:
    with path.open("rb") as handle:
        batch = pickle.load(handle, encoding="bytes")
    data = batch[b"data"]
    labels = np.array(batch[b"labels"], dtype=int)
    return data, labels


def load_data(seed: int, max_samples_per_class: int, image_size: int = 32) -> tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray]:
    rng = np.random.default_rng(seed)
    batch_files = [EXTRACTED_DIR / f"data_batch_{index}" for index in range(1, 6)]
    all_data: list[np.ndarray] = []
    all_labels: list[np.ndarray] = []
    for path in batch_files:
        data, labels = _load_batch(path)
        all_data.append(data)
        all_labels.append(labels)

    X = np.vstack(all_data)
    labels = np.hstack(all_labels)
    mask = (labels == CAT_LABEL) | (labels == NEG_LABEL)
    X = X[mask]
    labels = labels[mask]
    y = (labels == CAT_LABEL).astype(int)

    idx_cat = np.where(labels == CAT_LABEL)[0]
    idx_ship = np.where(labels == NEG_LABEL)[0]
    keep_count = min(max_samples_per_class, len(idx_cat), len(idx_ship))
    idx = np.concatenate(
        [
            rng.choice(idx_cat, size=keep_count, replace=False),
            rng.choice(idx_ship, size=keep_count, replace=False),
        ]
    )
    rng.shuffle(idx)
    X = X[idx].astype(np.float32) / 255.0
    y = y[idx]

    test_batch_file = EXTRACTED_DIR / "test_batch"
    data, labels = _load_batch(test_batch_file)
    X_test = data
    labels_test = labels
    mask_test = (labels_test == CAT_LABEL) | (labels_test == NEG_LABEL)
    X_test = X_test[mask_test]
    labels_test = labels_test[mask_test]
    y_test = (labels_test == CAT_LABEL).astype(int)

    idx_cat_test = np.where(labels_test == CAT_LABEL)[0]
    idx_ship_test = np.where(labels_test == NEG_LABEL)[0]
    keep_count_test = min(max_samples_per_class, len(idx_cat_test), len(idx_ship_test))
    idx_test = np.concatenate(
        [
            rng.choice(idx_cat_test, size=keep_count_test, replace=False),
            rng.choice(idx_ship_test, size=keep_count_test, replace=False),
        ]
    )
    rng.shuffle(idx_test)
    X_test = X_test[idx_test].astype(np.float32) / 255.0
    y_test = y_test[idx_test]

    X = X.reshape(-1, 3, 32, 32)
    X_test = X_test.reshape(-1, 3, 32, 32)
    if image_size != 32:
        t = torch.from_numpy(X)
        t = F.interpolate(t, size=(image_size, image_size), mode="bilinear", align_corners=False)
        X = t.numpy()
        t = torch.from_numpy(X_test)
        t = F.interpolate(t, size=(image_size, image_size), mode="bilinear", align_corners=False)
        X_test = t.numpy()

    X_train = torch.tensor(X, dtype=torch.float32)
    y_train = torch.tensor(y, dtype=torch.long)
    X_test = torch.tensor(X_test, dtype=torch.float32)
    y_test = torch.tensor(y_test, dtype=torch.long)

    X_train = TF.rgb_to_grayscale(X_train, num_output_channels=1)
    X_test = TF.rgb_to_grayscale(X_test, num_output_channels=1)

    train_ds = TensorDataset(X_train, y_train)
    test_ds = TensorDataset(X_test, y_test)
    return train_ds, test_ds

class Sampling(nn.Module):
    def forward(self, z_mean, z_log_var):
        epsilon = torch.randn_like(z_mean)
        return z_mean + torch.exp(0.5 * z_log_var) * epsilon

Loss, optimizer, and one training epoch

Implement vae_loss: unpack mu, logvar, and x_prime from the model output. The reconstruction term should compare x_prime to the input batch y_true using mean squared error (sum reduction, then divide by batch size so the scale is comparable across batch sizes $\frac{mse\_loss(x_{prime}, y_{true})}{batch\_size}$). The KL divergence from the prior uses the formula $-0.5 * \frac{\sum(1 + logvar - mu^2 - e^{logvar})}{batch\_size}$. The total loss is reconstruction plus BETA times KL.

Homework Assignment: Variational Autoencoder on CIFAR-10 (binary Cat vs. Ship)

In this exercise you will build a small convolutional VAE on grayscale images, train it with the ELBO objective, and inspect reconstructions before and after training. The notebook already downloads and subsamples CIFAR-10, builds DataLoaders, and fixes hyperparameters; your work is to fill in every # TODO so the model, loss, and training loop run end-to-end.

What you will practice

  • Implementing the encoder and decoder forward passes for a convolutional VAE
  • Wiring the encoder, sampling layer, and decoder in the VAE forward
  • Writing the reconstruction and KL terms of the loss (with the given BETA weight on KL)
  • A standard PyTorch optimization step inside the training loop
  • Calling your training and validation helpers from the epoch loop

What to expect The # TODO markers tell you exactly which expressions or statements are missing. Read the inline hints next to each TODO (for example the suggested F.mse_loss shape and the KL formula). If something fails, check tensor shapes against the comments on each conv layer (spatial size halves after each stride-2 block).

def _load_batch(path: pathlib.Path) -> tuple[np.ndarray, np.ndarray]:
    with path.open("rb") as handle:
        batch = pickle.load(handle, encoding="bytes")
    data = batch[b"data"]
    labels = np.array(batch[b"labels"], dtype=int)
    return data, labels


def load_data(seed: int, max_samples_per_class: int, image_size: int = 32) -> tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray]:
    rng = np.random.default_rng(seed)
    batch_files = [EXTRACTED_DIR / f"data_batch_{index}" for index in range(1, 6)]
    all_data: list[np.ndarray] = []
    all_labels: list[np.ndarray] = []
    for path in batch_files:
        data, labels = _load_batch(path)
        all_data.append(data)
        all_labels.append(labels)

    X = np.vstack(all_data)
    labels = np.hstack(all_labels)
    mask = (labels == CAT_LABEL) | (labels == NEG_LABEL)
    X = X[mask]
    labels = labels[mask]
    y = (labels == CAT_LABEL).astype(int)

    idx_cat = np.where(labels == CAT_LABEL)[0]
    idx_ship = np.where(labels == NEG_LABEL)[0]
    keep_count = min(max_samples_per_class, len(idx_cat), len(idx_ship))
    idx = np.concatenate(
        [
            rng.choice(idx_cat, size=keep_count, replace=False),
            rng.choice(idx_ship, size=keep_count, replace=False),
        ]
    )
    rng.shuffle(idx)
    X = X[idx].astype(np.float32) / 255.0
    y = y[idx]

    test_batch_file = EXTRACTED_DIR / "test_batch"
    data, labels = _load_batch(test_batch_file)
    X_test = data
    labels_test = labels
    mask_test = (labels_test == CAT_LABEL) | (labels_test == NEG_LABEL)
    X_test = X_test[mask_test]
    labels_test = labels_test[mask_test]
    y_test = (labels_test == CAT_LABEL).astype(int)

    idx_cat_test = np.where(labels_test == CAT_LABEL)[0]
    idx_ship_test = np.where(labels_test == NEG_LABEL)[0]
    keep_count_test = min(max_samples_per_class, len(idx_cat_test), len(idx_ship_test))
    idx_test = np.concatenate(
        [
            rng.choice(idx_cat_test, size=keep_count_test, replace=False),
            rng.choice(idx_ship_test, size=keep_count_test, replace=False),
        ]
    )
    rng.shuffle(idx_test)
    X_test = X_test[idx_test].astype(np.float32) / 255.0
    y_test = y_test[idx_test]

    X = X.reshape(-1, 3, 32, 32)
    X_test = X_test.reshape(-1, 3, 32, 32)
    if image_size != 32:
        t = torch.from_numpy(X)
        t = F.interpolate(t, size=(image_size, image_size), mode="bilinear", align_corners=False)
        X = t.numpy()
        t = torch.from_numpy(X_test)
        t = F.interpolate(t, size=(image_size, image_size), mode="bilinear", align_corners=False)
        X_test = t.numpy()

    X_train = torch.tensor(X, dtype=torch.float32)
    y_train = torch.tensor(y, dtype=torch.long)
    X_test = torch.tensor(X_test, dtype=torch.float32)
    y_test = torch.tensor(y_test, dtype=torch.long)

    X_train = TF.rgb_to_grayscale(X_train, num_output_channels=1)
    X_test = TF.rgb_to_grayscale(X_test, num_output_channels=1)

    train_ds = TensorDataset(X_train, y_train)
    test_ds = TensorDataset(X_test, y_test)
    return train_ds, test_ds

Encoder, decoder, and VAE (forward methods)

Complete the convolutional encoder: run the input through your three stride-2 convolution layers with a suitable nonlinearity between them, flatten the feature map, and pass the flattened vector through both fc_mean and fc_log_var to obtain z_mean and z_log_var. Then use the provided Sampling module to draw a latent vector z (reparameterization trick).

In [271]:
class Sampling(nn.Module):
    def forward(self, z_mean, z_log_var):
        epsilon = torch.randn_like(z_mean)
        return z_mean + torch.exp(0.5 * z_log_var) * epsilon


class Encoder(nn.Module):
    def __init__(self, image_size, embedding_dim):
        super().__init__()
        self.conv1 = nn.Conv2d(1, 32, 3, stride=2, padding=1)    # 64 -> 32
        self.conv2 = nn.Conv2d(32, 64, 3, stride=2, padding=1)   # 32 -> 16
        self.conv3 = nn.Conv2d(64, 128, 3, stride=2, padding=1)  # 16 -> 8
        self.flatten = nn.Flatten()
        self.fc_mean = nn.Linear(128 * (image_size // 8) * (image_size // 8), embedding_dim)
        self.fc_log_var = nn.Linear(128 * (image_size // 8) * (image_size // 8), embedding_dim)
        self.sampling = Sampling()

    def forward(self, x):
        x = F.relu(self.conv1(x))
        x = F.relu(self.conv2(x))
        x = F.relu(self.conv3(x))
        x = self.flatten(x)
        z_mean = self.fc_mean(x)
        z_log_var = self.fc_log_var(x)
        z = self.sampling(z_mean, z_log_var)
        return z_mean, z_log_var, z

Complete the decoder: apply fc, reshape the result to shape_before_flattening, run the transpose-convolution stack in reverse order of the encoder, and apply a final torch.sigmoid so pixel values lie in $[0, 1]$ to match the normalized images.

In [272]:
class Decoder(nn.Module):
    def __init__(self, embedding_dim, shape_before_flattening):
        super().__init__()
        self.shape_before_flattening = shape_before_flattening
        self.fc = nn.Linear(
            embedding_dim,
            shape_before_flattening[0] * shape_before_flattening[1] * shape_before_flattening[2],
        )
        self.deconv1 = nn.ConvTranspose2d(128, 64, 3, stride=2, padding=1, output_padding=1)  # 8 -> 16
        self.deconv2 = nn.ConvTranspose2d(64, 32, 3, stride=2, padding=1, output_padding=1)   # 16 -> 32
        self.deconv3 = nn.ConvTranspose2d(32, 1, 3, stride=2, padding=1, output_padding=1)    # 32 -> 64

    def forward(self, x):
        x = self.fc(x)
        x = x.view(-1, *self.shape_before_flattening)
        x = F.relu(self.deconv1(x))
        x = F.relu(self.deconv2(x))
        x = torch.sigmoid(self.deconv3(x))
        return x

Complete the VAE forward: call the encoder on x, then the decoder on the sampled z, and return the three tensors the loss function expects (z_mean, z_log_var, and the reconstruction).

In [273]:
class VAE(nn.Module):
    def __init__(self, encoder, decoder):
        super().__init__()
        self.encoder = encoder
        self.decoder = decoder

    def forward(self, x):
        z_mean, z_log_var, z = self.encoder(x)
        reconstruction = self.decoder(z)
        return z_mean, z_log_var, reconstruction

Loss, optimizer, and one training epoch

Implement vae_loss: unpack mu, logvar, and x_prime from the model output. The reconstruction term should compare x_prime to the input batch y_true using mean squared error (sum reduction, then divide by batch size so the scale is comparable across batch sizes $\frac{mse\_loss(x_{prime}, y_{true})}{batch\_size}$). The KL divergence from the prior uses the formula $-0.5 * \frac{\sum(1 + logvar - mu^2 - e^{logvar})}{batch\_size}$. The total loss is reconstruction plus BETA times KL.

In [275]:
def vae_loss(y_pred, y_true):
    mu, logvar, x_prime = y_pred
    recon_loss = F.mse_loss(x_prime, y_true, reduction="sum") / y_true.size(0)
    kl_loss = -0.5 * torch.sum(1 + logvar - mu.pow(2) - logvar.exp()) / y_true.size(0)
    return recon_loss + BETA * kl_loss

In train_one_epoch, for each batch zero gradients, compute loss from a forward pass, backpropagate, and step the optimizer. In validate, use the same loss function under torch.no_grad() and eval mode (already set on the VAE).

In [276]:
def train_one_epoch(vae, loader, optimizer):
    vae.train()
    running = 0.0
    for X, _ in tqdm(loader):
        X = X.to(DEVICE)
        optimizer.zero_grad()
        loss = vae_loss(vae(X), X)
        loss.backward()
        optimizer.step()
        running += loss.item() * X.size(0)
    return running / len(loader.dataset)


@torch.no_grad()
def validate(vae, loader):
    vae.eval()
    running = 0.0
    for X, _ in loader:
        X = X.to(DEVICE)
        loss = vae_loss(vae(X), X)
        running += loss.item() * X.size(0)
    return running / len(loader.dataset)

@torch.no_grad()
def recon_examples(vae, loader, n=8, title="Reconstructions"):
    vae.eval()
    x, y = next(iter(loader))
    x_device = x.to(next(vae.parameters()).device)

    z_mean, z_log_var, z = vae.encoder(x_device)
    recon = vae.decoder(z_mean).cpu()

    fig, axes = plt.subplots(n, 2, figsize=(4, 2 * n))
    if n == 1:
        axes = np.expand_dims(axes, axis=0)

    for i in range(n):
        axes[i, 0].imshow(x[i].squeeze(0), cmap="gray", vmin=0, vmax=1)
        axes[i, 0].set_title(f"Original: {CLASS_LABELS[int(y[i])]}")
        axes[i, 0].axis("off")

        axes[i, 1].imshow(recon[i].squeeze(0), cmap="gray", vmin=0, vmax=1)
        axes[i, 1].set_title(f"Recon")
        axes[i, 1].axis("off")

    plt.suptitle(title)
    plt.tight_layout()
    plt.show()

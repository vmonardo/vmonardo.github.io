def has_string_or_list(col):
    return any(isinstance(x, (str, np.ndarray)) for x in col)
mask = [not has_string_or_list(X[:, i]) for i in range(X.shape[1])]

def is_continuous(values: np.ndarray, min_unique: int = 20) -> bool:
    unique = np.unique(values[~np.isnan(values.astype(float))])
    return unique.size >= min_unique

feature_names = list(df_winsorized.columns[1:])

DISPLAY_NAMES = {
    "GrLivArea": "Above-Ground Living Area (sq ft)",
    "1stFlrSF": "First Floor Area (sq ft)",
    "2ndFlrSF": "Second Floor Area (sq ft)",

}


def display_name(raw: str) -> str:
    raw = raw.strip()
    if not raw:
        return ""
    if raw in DISPLAY_NAMES:
        return DISPLAY_NAMES[raw]

    replacements = {
        "gr": "above-ground",
        "liv": "living",
        "flr": "floor",
        "bsmt": "basement",
        "sf": "square feet",
        "yr": "year",
        "remod": "remodeled",
        "add": "added",
        "mas": "masonry",
        "vnr": "veneer",
        "tot": "total",
        "rms": "rooms",
        "abv": "above",
        "grd": "ground",
    }
    clean = raw.replace(".", " ").replace("_", " ").replace("-", " ")
    with_spaces = []
    prev_is_alpha = False
    prev_is_digit = False
    for ch in clean:
        is_alpha = ch.isalpha()
        is_digit = ch.isdigit()
        if (is_alpha and prev_is_digit) or (is_digit and prev_is_alpha):
            with_spaces.append(" ")
        with_spaces.append(ch)
        prev_is_alpha = is_alpha
        prev_is_digit = is_digit
    tokens = [t for t in "".join(with_spaces).split() if t]
    expanded = []
    for t in tokens:
        key = t.lower()
        expanded.append(replacements.get(key, t))
    out = " ".join([w if any(ch.isdigit() for ch in w) else w.title() for w in expanded])
    return out

def fit_linear_regression(X: np.ndarray, y: np.ndarray) -> np.ndarray:
    X_design = np.c_[np.ones(len(X)), X]
    beta, *_ = np.linalg.lstsq(X_design, y, rcond=None)
    return beta


def predict(X: np.ndarray, beta: np.ndarray) -> np.ndarray:
    X_design = np.c_[np.ones(len(X)), X]
    return X_design @ beta


def standardize(X: np.ndarray) -> Tuple[np.ndarray, np.ndarray, np.ndarray]:
    mean = X.mean(axis=0)
    std = X.std(axis=0, ddof=0)
    std[std == 0] = 1.0
    return (X - mean) / std, mean, std

X_train, X_test, y_train, y_test = train_test_split(np.array(X[:, mask], dtype=np.float32), y)

# Rank features by absolute coefficient magnitude (standardized)
beta_full = fit_linear_regression(X_train_std, y_train)
coef = beta_full[1:]
ranked = sorted(list(enumerate(coef)), key=lambda t: abs(t[1]), reverse=True)

# Keep only continuous, interpretable features for dropdown selection.
continuous_ranked = []
for idx, c in ranked:
    raw_name = feature_names[idx]
    name = raw_name.lower()
    if not is_continuous(np.array(X[:, mask], dtype=np.float32)[:, idx]):
        continue
    if "order" in name or "misc" in name:
        continue
    if not display_name(raw_name):
        continue
    continuous_ranked.append((idx, c))

if len(continuous_ranked) < 10:

# Demo: 1D models for each top feature
def fit_1d_line(x: np.ndarray, y_true: np.ndarray) -> Tuple[np.ndarray, np.ndarray]:
    X_design = np.c_[np.ones(len(x)), x]
    beta_1d, *_ = np.linalg.lstsq(X_design, y_true, rcond=None)
    y_pred = X_design @ beta_1d
    order = np.argsort(x)
    return x[order], y_pred[order]

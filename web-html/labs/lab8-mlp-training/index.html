Step 2. Define the MLP Model¶
An MLP is built by composing simple pieces:


    

  1. a linear map, z -> Wz + b
    2. 

  2. a nonlinear activation applied elementwise, such as ReLU
    3. 



For FashionMNIST, a 1-hidden-layer MLP looks like:


$x \in R^{784}$

$h = ReLU(W_1 x + b_1)$

logits = W_2 h + b_2, with one logit per class


The output is logits. We convert logits to probabilities only for interpretation.


















In [18]:






class SimpleMLP(nn.Module):
    def __init__(self, input_dim: int = 784, hidden_dim: int = 128, num_classes: int = 10):
        super().__init__()
        # Hyperparameters (chosen by us): input_dim, hidden_dim, num_classes
        self.fc1 = nn.Linear(input_dim, hidden_dim)   # W1, b1
        self.act = nn.ReLU()                          # nonlinearity (no learnable parameters)
        self.fc2 = nn.Linear(hidden_dim, num_classes) # W2, b2

    def forward(self, x):
        # x arrives as images: (batch, 1, 28, 28)
        # Flatten pixels into a feature vector: (batch, 784)
        x = x.view(x.size(0), -1)

        # Intermediate variables are useful for thinking/debugging:
        z1 = self.fc1(x)       # (batch, hidden_dim)
        h  = self.act(z1)      # (batch, hidden_dim)  elementwise nonlinearity
        logits = self.fc2(h)   # (batch, 10)
        return logits

Step 3. Loss Function and Optimizer¶
The model outputs logits. To train for multiclass classification, we want the model to put high probability on the correct class.


Softmax turns logits into probabilities, and cross-entropy penalizes the model when it assigns low probability to the true label.


In PyTorch, nn.CrossEntropyLoss() takes logits directly and computes a numerically stable version of log-softmax plus negative log-likelihood.

Step 4. Training Loop¶
A standard gradient-based training loop has four steps:


    

  1. Forward pass: compute logits f_theta(x)
    2. 

  2. Loss: measure how wrong the prediction is
    3. 

  3. Backward pass: autograd computes gradients
    4. 

  4. Update: the optimizer adjusts parameters to reduce loss
    5. 



We track average loss and accuracy for each epoch.


















In [22]:






def accuracy_from_logits(logits: torch.Tensor, y: torch.Tensor) -> float:
    preds = logits.argmax(dim=1)
    return (preds == y).float().mean().item()

def train_one_epoch(model, loader, loss_fn, optimizer, device):
    model.train()

    total_loss = 0.0
    total_correct = 0
    total_seen = 0

    # (Optional) track gradient magnitude to build intuition
    grad_norms = []

    for X, y in loader:
        X, y = X.to(device), y.to(device)

        optimizer.zero_grad(set_to_none=True)
        logits = model(X)
        loss = loss_fn(logits, y)
        loss.backward()

        # Compute an L2 norm of all gradients (one number per batch)
        total_norm_sq = 0.0
        for p in model.parameters():
            if p.grad is not None:
                total_norm_sq += p.grad.detach().data.norm(2).item() ** 2
        grad_norms.append(math.sqrt(total_norm_sq))

        optimizer.step()

        bs = y.size(0)
        total_loss += loss.item() * bs
        total_correct += (logits.argmax(dim=1) == y).sum().item()
        total_seen += bs

    avg_loss = total_loss / total_seen
    avg_acc = total_correct / total_seen
    avg_grad_norm = float(np.mean(grad_norms))
    return avg_loss, avg_acc, avg_grad_norm

@torch.no_grad()
def evaluate(model, loader, loss_fn, device):
    model.eval()

    total_loss = 0.0
    total_correct = 0
    total_seen = 0

    for X, y in loader:
        X, y = X.to(device), y.to(device)
        logits = model(X)
        loss = loss_fn(logits, y)

        bs = y.size(0)
        total_loss += loss.item() * bs
        total_correct += (logits.argmax(dim=1) == y).sum().item()
        total_seen += bs

    avg_loss = total_loss / total_seen
    avg_acc = total_correct / total_seen
    return avg_loss, avg_acc

@torch.no_grad()
def plot_softmax_before_after(model_trained, init_state, loader, example_index=0):
    # Recreate an identical model (same architecture) and load the saved *pre-training* weights
    model_before = SimpleMLP(hidden_dim=HIDDEN_DIM).to(device)
    model_before.load_state_dict(init_state)

    # Grab one batch and select a single example
    X, y = next(iter(loader))
    X, y = X.to(device), y.to(device)
    x0 = X[example_index:example_index+1]
    y0 = int(y[example_index].cpu())

    # Forward pass: logits -> softmax probabilities
    logits_before = model_before(x0).squeeze(0)
    logits_after  = model_trained(x0).squeeze(0)

    probs_before = torch.softmax(logits_before, dim=0).cpu().numpy()
    probs_after  = torch.softmax(logits_after, dim=0).cpu().numpy()

    # Show the image we're classifying
    plt.figure(figsize=(3, 3))
    img = (x0.detach().cpu().squeeze(0).squeeze(0) * 0.5 + 0.5)  # undo normalization
    plt.imshow(img, cmap="gray")
    plt.title(f"True label: {class_names[y0]}")
    plt.axis("off")
    plt.show()

    # Probability bar chart (grouped bars: before vs after)
    x = np.arange(len(class_names))
    width = 0.4

    plt.figure(figsize=(10, 4))
    plt.bar(x - width/2, probs_before, width, label="before training")
    plt.bar(x + width/2, probs_after,  width, label="after training")
    plt.xticks(x, class_names, rotation=35, ha="right")
    plt.ylabel("Softmax probability")
    plt.title("Softmax probabilities on one test example (before vs after training)")
    plt.legend()
    plt.tight_layout()
    plt.show()

plot_softmax_before_after(model, init_state, test_loader, example_index=0)

@torch.no_grad()
def show_predictions(model, loader, n_show=12):
    model.eval()
    X, y = next(iter(loader))
    X, y = X.to(device), y.to(device)

    logits = model(X)
    preds = logits.argmax(dim=1)

        pred_label = class_names[int(preds[i].cpu())]
        plt.title(f"pred: {pred_label}\ntrue: {true_label}", fontsize=9)
        plt.axis("off")
    plt.tight_layout()
    plt.show()

Step 7. Optional Confusion Matrix Evaluation¶
Accuracy tells us how often the model is correct overall, but it does not tell us what kinds of mistakes the model is making.


A confusion matrix gives a more detailed view of which true classes are being predicted as which other classes.






















Note: Reading a Confusion Matrix¶
For a classification problem with C classes, the confusion matrix is a C x C table.


    

  • Rows are true labels.
    • 

  • Columns are predicted labels.
    • 

  • Large values on the diagonal are correct predictions.
    • 

  • Large values off the diagonal are systematic mistakes.
    • 



The code below counts each (true, predicted) pair, visualizes the matrix as a heatmap, and prints the most common misclassifications.


















In [27]:






@torch.no_grad()
def confusion_matrix(model, loader, num_classes=10):
    model.eval()
    cm = np.zeros((num_classes, num_classes), dtype=np.int64)

    for X, y in loader:
        X = X.to(device)
        logits = model(X).cpu().numpy()
        preds = logits.argmax(axis=1)

        y_np = y.numpy()
        for t, p in zip(y_np, preds):
            cm[t, p] += 1
    return cm

cm = confusion_matrix(model, test_loader, num_classes=10)

topk = flat.argsort()[-8:][::-1]
print("Most common confusions (true -> pred):")
for idx in topk:

    p = idx % 10

Step 7. Optional Confusion Matrix Evaluation¶
Accuracy tells us how often the model is correct overall, but it does not tell us what kinds of mistakes the model is making.


A confusion matrix gives a more detailed view of which true classes are being predicted as which other classes.






















Note: Reading a Confusion Matrix¶
For a classification problem with C classes, the confusion matrix is a C x C table.


    

  • Rows are true labels.
    • 

  • Columns are predicted labels.
    • 

  • Large values on the diagonal are correct predictions.
    • 

  • Large values off the diagonal are systematic mistakes.
    • 



The code below counts each (true, predicted) pair, visualizes the matrix as a heatmap, and prints the most common misclassifications.


















In [ ]:






@torch.no_grad()
def confusion_matrix(model, loader, num_classes=10):
    model.eval()
    cm = np.zeros((num_classes, num_classes), dtype=np.int64)

    for X, y in loader:
        X = X.to(device)
        logits = model(X).cpu().numpy()
        preds = logits.argmax(axis=1)

        y_np = y.numpy()
        for t, p in zip(y_np, preds):
            cm[t, p] += 1
    return cm

cm = confusion_matrix(model, test_loader, num_classes=10)

plt.ylabel("true class")

plt.colorbar()

cm_off = cm.copy()

flat = cm_off.flatten()
topk = flat.argsort()[-8:][::-1]
print("Most common confusions (true -> pred):")
for idx in topk:
    t = idx // 10
    p = idx % 10
    if cm_off[t, p] > 0:
        print(f"  {t} -> {p}: {cm_off[t, p]}")
